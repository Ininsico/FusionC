NETWORK Office
DEVICE Server
SET threshold = 80
SET ratio = 2.5
SET total = (threshold + 20) * 2 - 10 / 5

SCAN Office
SCAN PORT 22

  ALERT "SSH Port Exposed"
  LOG "Security Warning"
END

IF PORT 443 CLOSED
  LOG "HTTPS is closed"
END

IF ATTACK DDOS DETECTED
  ALERT ADMIN
END

IF TRAFFIC >= threshold
  ALERT "He said \"high traffic\""
  BLOCK SOURCE
END

IF total != 5
  LOG "not five"
END
